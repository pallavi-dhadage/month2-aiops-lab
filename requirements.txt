# 1. Importing a Module (from the same directory)
import my_module

# 2. Importing from a Package (from the utils folder)
from utils import log_helper

print("=" * 40)
print("1. MODULES")
print("=" * 40)
# Using the function from my_module
print(my_module.greet("Pallavi"))

print("\n" + "=" * 40)
print("2. PACKAGES & MINI PROJECT")
print("=" * 40)
# Our sample log data
logs = [
    {"time": "10:00:05", "severity": "INFO", "msg": "User logged in"},
    {"time": "10:02:15", "severity": "WARNING", "msg": "Disk space low"},
    {"time": "10:05:30", "severity": "ERROR", "msg": "DB connection failed"},
    {"time": "10:10:00", "severity": "CRITICAL", "msg": "Server down"},
]

# Using helper functions from our custom package
critical_alerts = log_helper.filter_critical_logs(logs)
print(f"Critical Alerts: {len(critical_alerts)} found.")
for log in critical_alerts:
    print(f" - [{log['severity']}] {log['msg']}")

sorted_logs = log_helper.sort_logs_by_severity(logs)
print("\nLogs sorted by Severity (Highest to Lowest):")
for log in sorted_logs:
    print(f" - [{log['severity']:8}] {log['msg']}")
